function bridge_getTeamColorCode(colorData: integer) :: text:
    if {_colorData} is 11:
        return "&9" # Azul
    if {_colorData} is 14:
        return "&c" # Vermelho
    if {_colorData} is 5:
        return "&a" # Verde
    if {_colorData} is 4:
        return "&e" # Amarelo
    return "&7"

function bridge_getTeamName(colorData: integer) :: text:
    if {_colorData} is 11:
        return "&9Time Azul"
    if {_colorData} is 14:
        return "&cTime Vermelho"
    if {_colorData} is 5:
        return "&aTime Verde"
    if {_colorData} is 4:
        return "&eTime Amarelo"
    return "&7Time"

function bridge_getFormattedPlayerName(p: player) :: text:
    set {_u} to {_p}'s uuid
    set {_inst} to {bridge::player::%{_u}%::instance}
    if {_inst} is set:
        set {_team} to {bridge::player::%{_u}%::team}
        set {_arena} to {bridge::instance::%{_inst}%::arena}
        set {_cData} to {bridge::arena::%{_arena}%::team::%{_team}%::color}
        set {_code} to bridge_getTeamColorCode({_cData})
        return "%{_code}%%name of {_p}%"
    return "&f%name of {_p}%"

function bridge_sendActionBar(p: player, msg: text):
    if {_p} is online:
        send action bar colored {_msg} to {_p}

function bridge_coreTitle(p: player, titleText: text, subText: text, ticks: integer = 20):
    if {_p} is online:
        set {_t} to colored {_titleText}
        set {_s} to colored {_subText}
        {_p}.sendTitle({_t}, {_s})

function bridge_corePling(p: player, pitchValue: number):
    if {_p} is online:
        set {_loc} to {_p}.getLocation()
        {_p}.playSound({_loc}, Sound.NOTE_PLING, 1.0, {_pitchValue})

function bridge_buildCage(center: location, colorData: integer, inst: text):
    set {_w} to world of {_center}
    set {_cx} to floor(x-coordinate of {_center})
    set {_cy} to floor(y-coordinate of {_center})
    set {_cz} to floor(z-coordinate of {_center})

    loop 5 times:
        set {_y} to ({_cy} - 1) + (loop-value-1 - 1)
        loop 5 times:
            set {_x} to ({_cx} - 2) + (loop-value-2 - 1)
            loop 5 times:
                set {_z} to ({_cz} - 2) + (loop-value-3 - 1)
                
                set {_isOuter} to false
                if {_y} is ({_cy} - 1) or ({_cy} + 3):
                    set {_isOuter} to true
                if {_x} is ({_cx} - 2) or ({_cx} + 2):
                    set {_isOuter} to true
                if {_z} is ({_cz} - 2) or ({_cz} + 2):
                    set {_isOuter} to true

                if {_isOuter} is true:
                    set {_loc} to location at {_x}, {_y}, {_z} in {_w}
                    set {_b} to {_loc}.getBlock()
                    if {_b}.getType().getId() is 0:
                        {_b}.setType(Material.STAINED_GLASS)
                        {_b}.setData({_colorData})
                        add {_loc} to {bridge::instance::%{_inst}%::cage_blocks::*}

function bridge_removeCage(inst: text):
    loop {bridge::instance::%{_inst}%::cage_blocks::*}:
        set block at loop-value to air
    delete {bridge::instance::%{_inst}%::cage_blocks::*}

function bridge_coreColor(d: integer) :: object:
    if {_d} is 11:
        return Color.fromRGB(30, 144, 255)
    if {_d} is 14:
        return Color.fromRGB(220, 20, 60)
    if {_d} is 5:
        return Color.fromRGB(50, 205, 50)
    if {_d} is 4:
        return Color.fromRGB(255, 215, 0)
    return Color.fromRGB(220, 20, 60)

function bridge_giveLeatherArmor(p: player, colorData: integer):
    set {_color} to bridge_coreColor({_colorData})
    set {_inv} to {_p}.getInventory()
    {_inv}.setHelmet(null)

    set {_chest} to new ItemStack(Material.LEATHER_CHESTPLATE, 1)
    set {_metaC} to {_chest}.getItemMeta()
    {_metaC}.setColor({_color})
    {_metaC}.spigot().setUnbreakable(true)
    {_chest}.setItemMeta({_metaC})
    {_inv}.setChestplate({_chest})

    set {_legs} to new ItemStack(Material.LEATHER_LEGGINGS, 1)
    set {_metaL} to {_legs}.getItemMeta()
    {_metaL}.setColor({_color})
    {_metaL}.spigot().setUnbreakable(true)
    {_legs}.setItemMeta({_metaL})
    {_inv}.setLeggings({_legs})

    set {_boots} to new ItemStack(Material.LEATHER_BOOTS, 1)
    set {_metaB} to {_boots}.getItemMeta()
    {_metaB}.setColor({_color})
    {_metaB}.spigot().setUnbreakable(true)
    {_boots}.setItemMeta({_metaB})
    {_inv}.setBoots({_boots})

function bridge_giveKit(p: player, teamColorData: integer):
    clear {_p}'s inventory
    set {_p}'s gamemode to survival
    {_p}.setHealth(20.0)
    {_p}.setFoodLevel(20)
    {_p}.setFireTicks(0)
    {_p}.setFallDistance(0.0)

    remove regeneration from {_p}
    remove absorption from {_p}
    delete {bridge::player::%{_p}'s uuid%::gapple_eaten}
    delete {-bridge::player::%{_p}'s uuid%::last_attacker_name}
    delete {-bridge::player::%{_p}'s uuid%::last_attacker_time}

    set {_sword} to new ItemStack(Material.IRON_SWORD, 1)
    set {_mS} to {_sword}.getItemMeta()
    {_mS}.spigot().setUnbreakable(true)
    {_sword}.setItemMeta({_mS})
    set slot 0 of {_p}'s inventory to {_sword}

    set {_bow} to new ItemStack(Material.BOW, 1)
    set {_mB} to {_bow}.getItemMeta()
    {_mB}.spigot().setUnbreakable(true)
    {_bow}.setItemMeta({_mB})
    set slot 1 of {_p}'s inventory to {_bow}

    set {_pick} to new ItemStack(Material.DIAMOND_PICKAXE, 1)
    {_pick}.addUnsafeEnchantment(Enchantment.DIG_SPEED, 2)
    set {_mP} to {_pick}.getItemMeta()
    {_mP}.spigot().setUnbreakable(true)
    {_pick}.setItemMeta({_mP})
    set slot 2 of {_p}'s inventory to {_pick}

    set {_clay} to new ItemStack(Material.STAINED_CLAY, 64)
    {_clay}.setDurability({_teamColorData})
    set slot 3 of {_p}'s inventory to {_clay}
    set slot 4 of {_p}'s inventory to {_clay}

    set slot 5 of {_p}'s inventory to 8 golden apples
    set slot 8 of {_p}'s inventory to 1 arrow

    bridge_giveLeatherArmor({_p}, {_teamColorData})
    {_p}.updateInventory()

function bridge_fixVisibility(inst: text):
    wait 2 ticks
    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        set {_p1} to loop-value-1
        loop {bridge::instance::%{_inst}%::team::2::players::*}:
            set {_p2} to loop-value-2
            if {_p1} is online:
                if {_p2} is online:
                    {_p1}.hidePlayer({_p2})
                    {_p1}.showPlayer({_p2})
                    {_p2}.hidePlayer({_p1})
                    {_p2}.showPlayer({_p1})

            loop {bridge::instance::%{_inst}%::team::1::players::*}:
                if loop-value-2 is online:
                    bridge_coreTitle(loop-value-2, "%{_teamName}%", {_subMsg}, 20)
                    bridge_sendActionBar(loop-value-2, "&e&lPAUSA ATIVA: &f%{_pTime}%s restantes &7(Clique no chat para retomar)")

                if loop-value-2 is online:
                    bridge_coreTitle(loop-value-2, "%{_teamName}%", {_subMsg}, 20)
                    bridge_sendActionBar(loop-value-2, "&e&lPAUSA ATIVA: &f%{_pTime}%s restantes &7(Clique no chat para retomar)")

            if {_pTime} <= 0:
                bridge_resumeFromPause({_inst})

        # 3. CONGELAMENTO POR DESCONEXÃO (JOGADOR FORA DA PARTIDA)
        else if {bridge::instance::%{_inst}%::frozen} is true:
            if {bridge::instance::%{_inst}%::freeze_reason} is "DISCONNECT":
                remove 1 from {bridge::instance::%{_inst}%::freeze_timer}
                set {_fTime} to {bridge::instance::%{_inst}%::freeze_timer}
                set {_dcP} to {bridge::instance::%{_inst}%::freeze_player_name}

                    if loop-value-2 is online:
                        bridge_coreTitle(loop-value-2, "&c&lJOGADOR DESCONECTADO", {_subMsg}, 20)
                        bridge_sendActionBar(loop-value-2, "&c&lAGUARDANDO RECONEXÃO DE %{_dcP}%&c: &e%{_fTime}%s")

                    bridge_handleDisconnectTimeout({_inst})

        # 4. TEMPO REGULAMENTAR DE PARTIDA
        else if {_state} is "INGAME":
            remove 1 from {bridge::instance::%{_inst}%::time_remaining}
            if {bridge::instance::%{_inst}%::time_remaining} <= 0:
                set {bridge::instance::%{_inst}%::time_remaining} to 0
                bridge_handleTimeEnd({_inst})

function bridge_broadcastInstanceActionBar(inst: text, msg: text):
    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            bridge_sendActionBar(loop-value, {_msg})
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            bridge_sendActionBar(loop-value, {_msg})

function bridge_handleTimeEnd(inst: text):
    if {bridge::instance::%{_inst}%::state} is not "INGAME":
        stop

    if {bridge::instance::%{_inst}%::golden_goal_voted} is true:
        stop

    set {_s1} to {bridge::instance::%{_inst}%::team::1::score}
    set {_s2} to {bridge::instance::%{_inst}%::team::2::score}
    if {_s1} is not set:
        set {_s1} to 0
    if {_s2} is not set:
        set {_s2} to 0

    if {_s1} > {_s2}:
        bridge_endMatch({_inst}, 1)
    else if {_s2} > {_s1}:
        bridge_endMatch({_inst}, 2)
    else:
        set {bridge::instance::%{_inst}%::golden_goal_voted} to true
        bridge_startVote({_inst}, "GOL DE OURO", 30)

function bridge_startVote(inst: text, type: text, duration: integer, reqTeam: integer = 0):
    set {bridge::instance::%{_inst}%::voting::active} to true
    set {bridge::instance::%{_inst}%::voting::type} to {_type}
    set {bridge::instance::%{_inst}%::voting::timer} to {_duration}
    set {bridge::instance::%{_inst}%::voting::request_team} to {_reqTeam}

    delete {bridge::instance::%{_inst}%::voting::votes::*}

    set {_msgHeader} to "%{bridge::config::prefix}% &e&lNOVA VOTAÇÃO: &f%{_type}% &7(%{_duration}%s)"
    set {_msgAsk} to "%{bridge::config::prefix}% &eClique abaixo para votar:"

    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            send "" to loop-value
            send {_msgHeader} to loop-value
            send {_msgAsk} to loop-value
            execute console command "tellraw %loop-value% {""text"":"""", ""extra"":[{""text"":""   &a&l[VOTAR SIM]  "",""clickEvent"":{""action"":""run_command"",""value"":""/bridgevote sim""}},{""text"":""&c&l[VOTAR NÃO]"",""clickEvent"":{""action"":""run_command"",""value"":""/bridgevote nao""}}]}"
            send "" to loop-value
            bridge_corePling(loop-value, 1.5)

    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            send "" to loop-value
            send {_msgHeader} to loop-value
            send {_msgAsk} to loop-value
            execute console command "tellraw %loop-value% {""text"":"""", ""extra"":[{""text"":""   &a&l[VOTAR SIM]  "",""clickEvent"":{""action"":""run_command"",""value"":""/bridgevote sim""}},{""text"":""&c&l[VOTAR NÃO]"",""clickEvent"":{""action"":""run_command"",""value"":""/bridgevote nao""}}]}"
            send "" to loop-value
            bridge_corePling(loop-value, 1.5)

command /bridgevote [<text>]:
    executable by: players
    trigger:
        set {_u} to player's uuid
        if {bridge::player::%{_u}%::state} is not "INGAME":
            stop
        set {_inst} to {bridge::player::%{_u}%::instance}
        if {bridge::instance::%{_inst}%::voting::active} is not true:
            send "%{bridge::config::erro}% &cNenhuma votação em andamento." to player
            stop

        if arg-1 is "sim" or "s":
            set {bridge::instance::%{_inst}%::voting::votes::%{_u}%} to "SIM"
            send "%{bridge::config::sucesso}% &aVocê votou &fSIM&a!" to player
        else if arg-1 is "nao" or "não" or "n":
            set {bridge::instance::%{_inst}%::voting::votes::%{_u}%} to "NAO"
            send "%{bridge::config::sucesso}% &cVocê votou &fNÃO&c!" to player
        else:
            send "%{bridge::config::erro}% &cUse: /bridgevote <sim|nao>" to player
            stop

        bridge_checkVoteEarlyComplete({_inst})

function bridge_checkVoteEarlyComplete(inst: text):
    set {_totalOnline} to 0
    set {_totalVotes} to 0

    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            add 1 to {_totalOnline}
            if {bridge::instance::%{_inst}%::voting::votes::%loop-value's uuid%} is set:
                add 1 to {_totalVotes}

    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            add 1 to {_totalOnline}
            if {bridge::instance::%{_inst}%::voting::votes::%loop-value's uuid%} is set:
                add 1 to {_totalVotes}

    if {_totalVotes} >= {_totalOnline}:
        bridge_evaluateVote({_inst})

function bridge_evaluateVote(inst: text):
    if {bridge::instance::%{_inst}%::voting::active} is not true:
        stop

    set {_type} to {bridge::instance::%{_inst}%::voting::type}
    set {_reqTeam} to {bridge::instance::%{_inst}%::voting::request_team}
    delete {bridge::instance::%{_inst}%::voting::active}

    set {_t1Yes} to 0
    set {_t1Total} to 0
    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            add 1 to {_t1Total}
            if {bridge::instance::%{_inst}%::voting::votes::%loop-value's uuid%} is "SIM":
                add 1 to {_t1Yes}

    set {_t2Yes} to 0
    set {_t2Total} to 0
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            add 1 to {_t2Total}
            if {bridge::instance::%{_inst}%::voting::votes::%loop-value's uuid%} is "SIM":
                add 1 to {_t2Yes}

    set {_t1Pass} to false
    set {_t2Pass} to false

    if {_t1Total} > 0:
        if ({_t1Yes} / {_t1Total}) >= 0.5:
            set {_t1Pass} to true
    if {_t2Total} > 0:
        if ({_t2Yes} / {_t2Total}) >= 0.5:
            set {_t2Pass} to true

    # RESULTADO: GOL DE OURO
    if {_type} is "GOL DE OURO":
        if {_t1Pass} is true:
            if {_t2Pass} is true:
                set {bridge::instance::%{_inst}%::golden_goal} to true
                set {bridge::instance::%{_inst}%::time_remaining} to 0
                set {bridge::instance::%{_inst}%::state} to "INGAME"
                delete {bridge::instance::%{_inst}%::frozen}
                
                set {_msg} to "%{bridge::config::prefix}% &a&lGOL DE OURO APROVADO! &eO próximo ponto vence a partida!"
                bridge_broadcastInstanceActionBar({_inst}, "&a&lGOL DE OURO APROVADO!")
                loop {bridge::instance::%{_inst}%::team::1::players::*}:
                    if loop-value is online:
                        send {_msg} to loop-value
                        bridge_coreTitle(loop-value, "&b&lGOL DE OURO!", "&ePróximo gol vence!", 40)
                loop {bridge::instance::%{_inst}%::team::2::players::*}:
                    if loop-value is online:
                        send {_msg} to loop-value
                        bridge_coreTitle(loop-value, "&b&lGOL DE OURO!", "&ePróximo gol vence!", 40)
                stop

        set {_msg} to "%{bridge::config::prefix}% &cGol de Ouro rejeitado. Partida encerrada em EMPATE!"
        loop {bridge::instance::%{_inst}%::team::1::players::*}:
            if loop-value is online:
                send {_msg} to loop-value
        loop {bridge::instance::%{_inst}%::team::2::players::*}:
            if loop-value is online:
                send {_msg} to loop-value
        bridge_endMatch({_inst}, 0)

    # RESULTADO: PAUSA (SOMENTE AGORA DEBITA O PAUSE USADO)
    else if {_type} is "PAUSA":
        if {_t1Pass} is true:
            if {_t2Pass} is true:
                # DEBITA A PAUSA UTILIZADA APENAS COM APROVAÇÃO
                add 1 to {bridge::instance::%{_inst}%::pauses::%{_reqTeam}%}

                set {_arena} to {bridge::instance::%{_inst}%::arena}
                set {_c1} to {bridge::arena::%{_arena}%::team::1::cage}
                if {_c1} is set:
                    set {_orig1} to {_c1}
                else:
                    set {_sponge1} to {bridge::arena::%{_arena}%::team::1::sponge}
                    set {_orig1} to location 7 blocks above {_sponge1}

                set {_c2} to {bridge::arena::%{_arena}%::team::2::cage}
                if {_c2} is set:
                    set {_orig2} to {_c2}
                else:
                    set {_sponge2} to {bridge::arena::%{_arena}%::team::2::sponge}
                    set {_orig2} to location 7 blocks above {_sponge2}

                set {_dest1} to bridge_getLoc({_orig1}, {_inst})
                set {_dest2} to bridge_getLoc({_orig2}, {_inst})

                set {_cData1} to {bridge::arena::%{_arena}%::team::1::color}
                set {_cData2} to {bridge::arena::%{_arena}%::team::2::color}

                bridge_buildCage({_dest1}, {_cData1}, {_inst})
                bridge_buildCage({_dest2}, {_cData2}, {_inst})

                loop {bridge::instance::%{_inst}%::team::1::players::*}:
                    if loop-value is online:
                        teleport loop-value to {_dest1}
                loop {bridge::instance::%{_inst}%::team::2::players::*}:
                    if loop-value is online:
                        teleport loop-value to {_dest2}

                set {bridge::instance::%{_inst}%::in_pause} to true
                set {bridge::instance::%{_inst}%::pause_request_team} to {_reqTeam}
                set {bridge::instance::%{_inst}%::pause_timer} to 30
                
                set {_msg} to "%{bridge::config::prefix}% &a&lPAUSA APROVADA! &ePartida congelada por 30s na cage."
                loop {bridge::instance::%{_inst}%::team::1::players::*}:
                    if loop-value is online:
                        send {_msg} to loop-value
                        execute console command "tellraw %loop-value% {""text"":"""", ""extra"":[{""text"":""&a&l[CLIQUE AQUI PARA RETOMAR PARTIDA]"",""clickEvent"":{""action"":""run_command"",""value"":""/despausar""}}]}"
                loop {bridge::instance::%{_inst}%::team::2::players::*}:
                    if loop-value is online:
                        send {_msg} to loop-value
                        execute console command "tellraw %loop-value% {""text"":"""", ""extra"":[{""text"":""&a&l[CLIQUE AQUI PARA RETOMAR PARTIDA]"",""clickEvent"":{""action"":""run_command"",""value"":""/despausar""}}]}"
                stop

        set {_msg} to "%{bridge::config::prefix}% &cSolicitação de pausa REJEITADA por falta de consenso entre os times."
        loop {bridge::instance::%{_inst}%::team::1::players::*}:
            if loop-value is online:
                send {_msg} to loop-value
        loop {bridge::instance::%{_inst}%::team::2::players::*}:
            if loop-value is online:
                send {_msg} to loop-value

# ---------------------------------------------------------
# AUTO-RECONEXÃO NO JOIN
# ---------------------------------------------------------
on join:
    wait 2 ticks
    set {_u} to player's uuid
    delete {_activeInst}
    
    loop {bridge::core::active_instances::*}:
        set {_inst} to loop-value-1
        loop {bridge::instance::%{_inst}%::team::1::players::*}:
            set {_pCheck} to loop-value-2
            if {_pCheck}'s uuid is {_u}:
                set {_activeInst} to {_inst}
                set {_activeTeam} to 1
                exit loop
        if {_activeInst} is not set:
            loop {bridge::instance::%{_inst}%::team::2::players::*}:

function bridge_reconnectPlayer(p: player, inst: text, team: integer):
    set {_u} to {_p}'s uuid
    set {bridge::player::%{_u}%::instance} to {_inst}
    set {bridge::player::%{_u}%::team} to {_team}
    set {bridge::player::%{_u}%::state} to "INGAME"

    set {_arena} to {bridge::instance::%{_inst}%::arena}
    set {_orig} to {bridge::arena::%{_arena}%::team::%{_team}%::sponge}
    set {_dest} to bridge_getLoc({_orig}, {_inst})
    
    teleport {_p} to {_dest}
    bridge_giveKit({_p}, {bridge::arena::%{_arena}%::team::%{_team}%::color})

    send "%{bridge::config::prefix}% &a&lRECONECTADO! &eVocê voltou para a partida!" to {_p}
    
    if {bridge::instance::%{_inst}%::frozen} is true:
        if {bridge::instance::%{_inst}%::freeze_reason} is "DISCONNECT":
            bridge_resumeMatchSpot({_inst})

function bridge_joinQueue(p: player, arena: text):
    delete {_opponent}
    loop {bridge::queue::%{_arena}%::*}:
        set {_candidate} to loop-value
        if {_candidate} is online:
            if {_candidate} is not {_p}:
                if {bridge::player::%{_candidate}'s uuid%::state} is "QUEUE":
                    if {bridge::player::%{_candidate}'s uuid%::queue_arena} is {_arena}:
                        set {_opponent} to {_candidate}
                        remove {_candidate} from {bridge::queue::%{_arena}%::*}
                        exit loop
        remove {_candidate} from {bridge::queue::%{_arena}%::*}
    if {_opponent} is set:
        bridge_startMatch({_opponent}, {_p}, {_arena})
        stop
    add {_p} to {bridge::queue::%{_arena}%::*}
    set {bridge::player::%{_p}'s uuid%::state} to "QUEUE"
    set {bridge::player::%{_p}'s uuid%::queue_arena} to {_arena}
    send "%{bridge::config::prefix}% &aVocê entrou na fila de &f%{_arena}%&a! Procurando oponente... &7(/l para sair)" to {_p}

function bridge_findFreeInstance(arena: text) :: text:
    loop {bridge::arena::%{_arena}%::instances::*}:
        set {_inst} to loop-value
        if {bridge::instances::free::*} contains {_inst}:
            set {_state} to {bridge::instance::%{_inst}%::state}
            if {_state} is not set:
                return {_inst}
            if {_state} is "IDLE":
                return {_inst}
    return ""

function bridge_startMatch(p1: player, p2: player, arena: text):
    set {_inst} to bridge_findFreeInstance({_arena})
    if {_inst} is "":
        bridge_expandInstances({_arena}, 1)
        set {_inst} to bridge_findFreeInstance({_arena})
    if {_inst} is "":
        remove {_p1} from {bridge::queue::%{_arena}%::*}
        remove {_p2} from {bridge::queue::%{_arena}%::*}
        delete {bridge::player::%{_p1}'s uuid%::*}
        delete {bridge::player::%{_p2}'s uuid%::*}
        send "%{bridge::config::erro}% &cFalha ao alocar arena. Digite /jogar novamente." to {_p1}
        send "%{bridge::config::erro}% &cFalha ao alocar arena. Digite /jogar novamente." to {_p2}
        stop

    remove {_inst} from {bridge::instances::free::*}
    if {bridge::core::active_instances::*} does not contain {_inst}:
        add {_inst} to {bridge::core::active_instances::*}

    set {bridge::instance::%{_inst}%::state} to "STARTING"
    
    set {_maxTime} to {bridge::arena::%{_arena}%::tempo}
    if {_maxTime} is not set:
        set {_maxTime} to 900
    set {bridge::instance::%{_inst}%::time_remaining} to {_maxTime}
    delete {bridge::instance::%{_inst}%::frozen}
    delete {bridge::instance::%{_inst}%::in_pause}
    delete {bridge::instance::%{_inst}%::golden_goal}
    delete {bridge::instance::%{_inst}%::golden_goal_voted}
    delete {bridge::instance::%{_inst}%::pauses::1}
    delete {bridge::instance::%{_inst}%::pauses::2}
    delete {bridge::instance::%{_inst}%::voting::active}
    
    if {bridge::core::session_counter} is not set:
        set {bridge::core::session_counter} to 0
    add 1 to {bridge::core::session_counter}
    set {bridge::instance::%{_inst}%::match_id} to {bridge::core::session_counter}
    
    delete {bridge::instance::%{_inst}%::closing_id}
    delete {bridge::instance::%{_inst}%::team::1::players::*}
    delete {bridge::instance::%{_inst}%::team::2::players::*}
    delete {bridge::instance::%{_inst}%::placed_blocks::*}

    add {_p1} to {bridge::instance::%{_inst}%::team::1::players::*}
    add {_p2} to {bridge::instance::%{_inst}%::team::2::players::*}

    set {bridge::instance::%{_inst}%::team::1::score} to 0
    set {bridge::instance::%{_inst}%::team::2::score} to 0

    delete {bridge::player::%{_p1}'s uuid%::queue_arena}
    delete {bridge::player::%{_p2}'s uuid%::queue_arena}

    set {bridge::player::%{_p1}'s uuid%::instance} to {_inst}
    set {bridge::player::%{_p1}'s uuid%::team} to 1
    set {bridge::player::%{_p1}'s uuid%::state} to "INGAME"
    set {bridge::player::%{_p1}'s uuid%::match_kills} to 0
    set {bridge::player::%{_p1}'s uuid%::match_goals} to 0

    set {bridge::player::%{_p2}'s uuid%::instance} to {_inst}
    set {bridge::player::%{_p2}'s uuid%::team} to 2
    set {bridge::player::%{_p2}'s uuid%::state} to "INGAME"
    set {bridge::player::%{_p2}'s uuid%::match_kills} to 0
    set {bridge::player::%{_p2}'s uuid%::match_goals} to 0

    set {_n1} to bridge_getFormattedPlayerName({_p1})
    set {_n2} to bridge_getFormattedPlayerName({_p2})

    send "%{bridge::config::prefix}% &aPartida encontrada! Você jogará contra %{_n2}%&a." to {_p1}
    send "%{bridge::config::prefix}% &aPartida encontrada! Você jogará contra %{_n1}%&a." to {_p2}

    bridge_startRound({_inst})

function bridge_startRound(inst: text, goalScorerName: text = ""):
    set {_state} to {bridge::instance::%{_inst}%::state}
    if {_state} is not "STARTING":
        if {_state} is not "GOAL":
            stop

    set {_matchId} to {bridge::instance::%{_inst}%::match_id}
    if {_matchId} is not set:
        stop

    set {_arena} to {bridge::instance::%{_inst}%::arena}
    set {bridge::instance::%{_inst}%::state} to "CAGE"

    set {_c1} to {bridge::arena::%{_arena}%::team::1::cage}
    if {_c1} is set:
        set {_orig1} to {_c1}
    else:
        set {_sponge1} to {bridge::arena::%{_arena}%::team::1::sponge}
        set {_orig1} to location 7 blocks above {_sponge1}
        set yaw of {_orig1} to yaw of {_sponge1}
        set pitch of {_orig1} to pitch of {_sponge1}

    set {_c2} to {bridge::arena::%{_arena}%::team::2::cage}
    if {_c2} is set:
        set {_orig2} to {_c2}
    else:
        set {_sponge2} to {bridge::arena::%{_arena}%::team::2::sponge}
        set {_orig2} to location 7 blocks above {_sponge2}
        set yaw of {_orig2} to yaw of {_sponge2}
        set pitch of {_orig2} to pitch of {_sponge2}

    set {_dest1} to bridge_getLoc({_orig1}, {_inst})
    set {_dest2} to bridge_getLoc({_orig2}, {_inst})

    set {_cData1} to {bridge::arena::%{_arena}%::team::1::color}
    set {_cData2} to {bridge::arena::%{_arena}%::team::2::color}
    set {_tCode1} to bridge_getTeamColorCode({_cData1})
    set {_tCode2} to bridge_getTeamColorCode({_cData2})

    bridge_buildCage({_dest1}, {_cData1}, {_inst})
    bridge_buildCage({_dest2}, {_cData2}, {_inst})

    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            teleport loop-value to {_dest1}
            bridge_giveKit(loop-value, {_cData1})
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            teleport loop-value to {_dest2}
            bridge_giveKit(loop-value, {_cData2})

    bridge_fixVisibility({_inst})

    set {_s1} to {bridge::instance::%{_inst}%::team::1::score}
    set {_s2} to {bridge::instance::%{_inst}%::team::2::score}

    loop 5 times:
        if {bridge::instance::%{_inst}%::match_id} is not {_matchId}:
            bridge_removeCage({_inst})
            stop
        if {bridge::instance::%{_inst}%::state} is not "CAGE":
            bridge_removeCage({_inst})
            stop

        set {_sec} to 6 - loop-value-1
        set {_sub} to "&7A rodada vai começar!"
        if {_goalScorerName} is not "":
            set {_sub} to "&fPlacar: %{_tCode1}%%{_s1}% &8x %{_tCode2}%%{_s2}% &7(Iniciando em %{_sec}%s)"

        loop {bridge::instance::%{_inst}%::team::1::players::*}:
            if loop-value-2 is online:
                if {_goalScorerName} is not "":
                    bridge_coreTitle(loop-value-2, "&aGOL DE %{_goalScorerName}%!", {_sub}, 20)
                else:
                    bridge_coreTitle(loop-value-2, "&e%{_sec}%", {_sub}, 20)
                bridge_corePling(loop-value-2, 1.0)
                if {_sec} is 5:
                    if {_goalScorerName} is not "":
                        send "%{bridge::config::prefix}% &aGOL DE %{_goalScorerName}%! &fPlacar: %{_tCode1}%%{_s1}% &8x %{_tCode2}%%{_s2}%" to loop-value-2
                    else:
                        send "%{bridge::config::prefix}% &eA rodada inicia em 5 segundos..." to loop-value-2

        loop {bridge::instance::%{_inst}%::team::2::players::*}:
            if loop-value-2 is online:
                if {_goalScorerName} is not "":
                    bridge_coreTitle(loop-value-2, "&aGOL DE %{_goalScorerName}%!", {_sub}, 20)
                else:
                    bridge_coreTitle(loop-value-2, "&e%{_sec}%", {_sub}, 20)
                bridge_corePling(loop-value-2, 1.0)
                if {_sec} is 5:
                    if {_goalScorerName} is not "":
                        send "%{bridge::config::prefix}% &aGOL DE %{_goalScorerName}%! &fPlacar: %{_tCode1}%%{_s1}% &8x %{_tCode2}%%{_s2}%" to loop-value-2
                    else:
                        send "%{bridge::config::prefix}% &eA rodada inicia em 5 segundos..." to loop-value-2

        wait 1 second

    if {bridge::instance::%{_inst}%::match_id} is not {_matchId}:
        bridge_removeCage({_inst})
        stop
    if {bridge::instance::%{_inst}%::state} is not "CAGE":
        bridge_removeCage({_inst})
        stop

    bridge_removeCage({_inst})
    set {bridge::instance::%{_inst}%::state} to "INGAME"

    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            bridge_coreTitle(loop-value, "&aVALENDO!", "&7Cruze a ponte e marque o gol!", 20)
            send "%{bridge::config::prefix}% &a&lVALENDO! &7Cruze a ponte e marque o gol!" to loop-value
            bridge_corePling(loop-value, 2.0)
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            bridge_coreTitle(loop-value, "&aVALENDO!", "&7Cruze a ponte e marque o gol!", 20)
            send "%{bridge::config::prefix}% &a&lVALENDO! &7Cruze a ponte e marque o gol!" to loop-value
            bridge_corePling(loop-value, 2.0)

# ---------------------------------------------------------
function bridge_respawnPlayer(p: player, deathType: text = "VOID", killerName: text = ""):
    set {_inst} to {bridge::player::%{_p}'s uuid%::instance}
    if {_inst} is not set:
        stop
    set {_arena} to {bridge::instance::%{_inst}%::arena}
    set {_team} to {bridge::player::%{_p}'s uuid%::team}
    set {_orig} to {bridge::arena::%{_arena}%::team::%{_team}%::sponge}
    set {_dest} to bridge_getLoc({_orig}, {_inst})
    
    set {_vNameFormatted} to bridge_getFormattedPlayerName({_p})

    if {_deathType} is not "SELF_PORTAL":
        bridge_addStat({_p}, "deaths", 1)
        if {_killerName} is not "":
            set {_pKiller} to {_killerName} parsed as offline player
            if {_pKiller} is online:
                bridge_addStat({_pKiller}, "kills", 1)
                add 1 to {bridge::player::%{_pKiller}'s uuid%::match_kills}

    if {_killerName} is not "":
        set {_pKillerObj} to {_killerName} parsed as offline player

function bridge_scoreGoal(inst: text, p: player, team: integer):
    if {bridge::instance::%{_inst}%::state} is not "INGAME":
        stop

    set {bridge::instance::%{_inst}%::state} to "GOAL"
    add 1 to {bridge::instance::%{_inst}%::team::%{_team}%::score}
    
    bridge_addStat({_p}, "goals", 1)
    add 1 to {bridge::player::%{_p}'s uuid%::match_goals}

    set {_arena} to {bridge::instance::%{_inst}%::arena}
    set {_scorerFormatted} to bridge_getFormattedPlayerName({_p})
    set {_s1} to {bridge::instance::%{_inst}%::team::1::score}
    set {_s2} to {bridge::instance::%{_inst}%::team::2::score}

    set {_tCode1} to bridge_getTeamColorCode({bridge::arena::%{_arena}%::team::1::color})
    set {_tCode2} to bridge_getTeamColorCode({bridge::arena::%{_arena}%::team::2::color})
    set {_msgGoal} to "%{bridge::config::prefix}% &a&lGOL! %{_scorerFormatted}% &7marcou o ponto! &8(%{_tCode1}%%{_s1}% &8x %{_tCode2}%%{_s2}%&8)"

    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            send {_msgGoal} to loop-value
            {_p}.playSound({_p}.getLocation(), Sound.LEVEL_UP, 1.0, 1.0)
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            send {_msgGoal} to loop-value
            {_p}.playSound({_p}.getLocation(), Sound.LEVEL_UP, 1.0, 1.0)

    if {bridge::instance::%{_inst}%::golden_goal} is true:
        bridge_endMatch({_inst}, {_team})
    else if {bridge::instance::%{_inst}%::team::%{_team}%::score} >= {bridge::arena::%{_arena}%::gols}:
        bridge_endMatch({_inst}, {_team})
    else:
        bridge_startRound({_inst}, {_scorerFormatted})

        if damage cause is fall:
            cancel event
            stop

        if attacker is player:
            if attacker is not victim:
                set {-bridge::player::%victim's uuid%::last_attacker_name} to name of attacker
                set {-bridge::player::%victim's uuid%::last_attacker_time} to (unix timestamp of now)

        if damage cause is void:

            set {_lastAtkName} to {-bridge::player::%victim's uuid%::last_attacker_name}
            set {_lastAtkTime} to {-bridge::player::%victim's uuid%::last_attacker_time}

function bridge_checkDisconnectState(inst: text, team: integer, dcName: text):
    set {_onlineCount} to 0
    set {_totalCount} to size of {bridge::instance::%{_inst}%::team::%{_team}%::players::*}
    loop {bridge::instance::%{_inst}%::team::%{_team}%::players::*}:
        if loop-value is online:
            add 1 to {_onlineCount}

    if {_totalCount} > 0:
        if ({_onlineCount} / {_totalCount}) >= 0.5:
            set {_msg} to "%{bridge::config::prefix}% &c%{_dcName}% &cdesconectou. A partida continua!"
            loop {bridge::instance::%{_inst}%::team::1::players::*}:
                if loop-value is online:
                    send {_msg} to loop-value
            loop {bridge::instance::%{_inst}%::team::2::players::*}:
                if loop-value is online:
                    send {_msg} to loop-value
            stop

    set {bridge::instance::%{_inst}%::frozen} to true
    set {bridge::instance::%{_inst}%::freeze_reason} to "DISCONNECT"
    set {bridge::instance::%{_inst}%::freeze_player_name} to {_dcName}
    set {bridge::instance::%{_inst}%::freeze_timer} to 30

    set {_msg} to "%{bridge::config::prefix}% &c&lJOGADOR DESCONECTADO! &e%{_dcName}% &7saiu. Partida congelada no local (&f30s&7)."
    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            send {_msg} to loop-value
            bridge_coreTitle(loop-value, "&c&lJOGADOR DESCONECTADO", "&7Aguardando %{_dcName}%", 40)
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            send {_msg} to loop-value
            bridge_coreTitle(loop-value, "&c&lJOGADOR DESCONECTADO", "&7Aguardando %{_dcName}%", 40)

function bridge_handleDisconnectTimeout(inst: text):
    set {_winner} to 1
    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            set {_winner} to 1
            exit loop
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            set {_winner} to 2
            exit loop

    set {_msgWO} to "%{bridge::config::prefix}% &cTempo de reconexão esgotado! Vitória por W.O.!"
    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            send {_msgWO} to loop-value
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            send {_msgWO} to loop-value

    bridge_endMatch({_inst}, {_winner})

function bridge_endMatch(inst: text, winnerTeam: integer):
    set {_state} to {bridge::instance::%{_inst}%::state}
    set {_canEnd} to false
    if {_state} is "STARTING":
        set {_canEnd} to true
    if {_state} is "CAGE":
        set {_canEnd} to true
    if {_state} is "INGAME":
        set {_canEnd} to true
    if {_state} is "GOAL":
        set {_canEnd} to true

    if {_canEnd} is false:
        stop

    set {_closingId} to {bridge::instance::%{_inst}%::match_id}
    if {_closingId} is not set:
        stop

    set {bridge::instance::%{_inst}%::state} to "CELEBRATION"
    set {bridge::instance::%{_inst}%::closing_id} to {_closingId}
    delete {bridge::instance::%{_inst}%::match_id}

    remove {_inst} from {bridge::core::active_instances::*}
    bridge_removeCage({_inst})

    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        bridge_addStat(loop-value, "games", 1)
        if {_winnerTeam} is 1:
            bridge_addStat(loop-value, "wins", 1)
        else if {_winnerTeam} is 2:
            bridge_addStat(loop-value, "losses", 1)

    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        bridge_addStat(loop-value, "games", 1)
        if {_winnerTeam} is 2:
            bridge_addStat(loop-value, "wins", 1)
        else if {_winnerTeam} is 1:
            bridge_addStat(loop-value, "losses", 1)

    set {_arena} to {bridge::instance::%{_inst}%::arena}
    if {_winnerTeam} > 0:
        set {_winColorData} to {bridge::arena::%{_arena}%::team::%{_winnerTeam}%::color}
        set {_winTeamName} to bridge_getTeamName({_winColorData})
        set {_msgEnd} to "%{bridge::config::prefix}% &a&lFIM DE PARTIDA! &7O %{_winTeamName}% &7venceu o jogo!"
        set {_subEnd} to "&7O %{_winTeamName}% &7venceu a partida!"
    else:
        set {_msgEnd} to "%{bridge::config::prefix}% &e&lFIM DE PARTIDA! &7A partida terminou em &e&lEMPATE&7!"
        set {_subEnd} to "&7A partida terminou em EMPATE!"

    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        if loop-value is online:
            bridge_coreTitle(loop-value, "&aFIM DE PARTIDA!", {_subEnd}, 60)
            send {_msgEnd} to loop-value
    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        if loop-value is online:
            bridge_coreTitle(loop-value, "&aFIM DE PARTIDA!", {_subEnd}, 60)
            send {_msgEnd} to loop-value

    wait 4 seconds

    if {bridge::instance::%{_inst}%::closing_id} is not {_closingId}:
        stop

    set {bridge::instance::%{_inst}%::state} to "RESETTING"
    set {_lobby} to {bridge::config::lobby}

    loop {bridge::instance::%{_inst}%::team::1::players::*}:
        set {_p} to loop-value
        if {bridge::player::%{_p}'s uuid%::instance} is {_inst}:
            if {_p} is online:
                clear {_p}'s inventory
                {_p}.setHealth(20.0)
                {_p}.setFoodLevel(20)
                {_p}.setFallDistance(0)
                set {_zero} to new Vector(0, 0, 0)
                {_p}.setVelocity({_zero})
                
                remove regeneration from {_p}
                remove absorption from {_p}
                
                set {_inv} to {_p}.getInventory()
                {_inv}.setHelmet(null)
                {_inv}.setChestplate(null)
                {_inv}.setLeggings(null)
                {_inv}.setBoots(null)
                
                teleport {_p} to {_lobby}
                set {_p}'s gamemode to adventure
            delete {bridge::player::%{_p}'s uuid%::match_kills}
            delete {bridge::player::%{_p}'s uuid%::match_goals}
            delete {bridge::player::%{_p}'s uuid%::*}

    loop {bridge::instance::%{_inst}%::team::2::players::*}:
        set {_p} to loop-value
        if {bridge::player::%{_p}'s uuid%::instance} is {_inst}:
            if {_p} is online:
                clear {_p}'s inventory
                {_p}.setHealth(20.0)
                {_p}.setFoodLevel(20)
                {_p}.setFallDistance(0)
                set {_zero} to new Vector(0, 0, 0)
                {_p}.setVelocity({_zero})
                
                remove regeneration from {_p}
                remove absorption from {_p}
                
                set {_inv} to {_p}.getInventory()
                {_inv}.setHelmet(null)
                {_inv}.setChestplate(null)
                {_inv}.setLeggings(null)
                {_inv}.setBoots(null)
                
                teleport {_p} to {_lobby}
                set {_p}'s gamemode to adventure
            delete {bridge::player::%{_p}'s uuid%::match_kills}
            delete {bridge::player::%{_p}'s uuid%::match_goals}
            delete {bridge::player::%{_p}'s uuid%::*}

    delete {bridge::instance::%{_inst}%::time_remaining}
    delete {bridge::instance::%{_inst}%::frozen}
    delete {bridge::instance::%{_inst}%::in_pause}
    delete {bridge::instance::%{_inst}%::pause_request_team}
    delete {bridge::instance::%{_inst}%::pause_timer}
    delete {bridge::instance::%{_inst}%::freeze_reason}
    delete {bridge::instance::%{_inst}%::freeze_player_name}
    delete {bridge::instance::%{_inst}%::freeze_timer}
    delete {bridge::instance::%{_inst}%::golden_goal}
    delete {bridge::instance::%{_inst}%::golden_goal_voted}
    delete {bridge::instance::%{_inst}%::pauses::1}
    delete {bridge::instance::%{_inst}%::pauses::2}
    delete {bridge::instance::%{_inst}%::voting::*}
    delete {bridge::instance::%{_inst}%::team::1::players::*}
    delete {bridge::instance::%{_inst}%::team::2::players::*}
    delete {bridge::instance::%{_inst}%::placed_blocks::*}

    set {_ok} to bridge_resetInstance({_inst})
    delete {bridge::instance::%{_inst}%::closing_id}

    if {_ok} is true:
        set {bridge::instance::%{_inst}%::state} to "IDLE"
    else:
        remove {_inst} from {bridge::instances::free::*}
        set {bridge::instance::%{_inst}%::state} to "DISABLED"

on quit:
    set {_uuid} to player's uuid
    if {bridge::player::%{_uuid}%::state} is "QUEUE":
        set {_arena} to {bridge::player::%{_uuid}%::queue_arena}
