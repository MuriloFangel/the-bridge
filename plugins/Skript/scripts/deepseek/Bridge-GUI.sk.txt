import:
    org.bukkit.Bukkit
    org.bukkit.Material
    org.bukkit.inventory.ItemStack
    java.util.Arrays

command /arenas:
    aliases: /menu, /playmenu
    executable by: players
    trigger:
        bridge_openArenaMenu(player)

function bridge_openArenaMenu(p: player):
    set {_inv} to Bukkit.createInventory(null, 27, colored "&8The Bridge - Seleção de Arenas")

    set {_slot} to 10
    loop {bridge::arenas::*}:
        set {_arena} to loop-value
        set {_freeCount} to 0
        
        loop {bridge::arena::%{_arena}%::instances::*}:
            if {bridge::instances::free::*} contains loop-value-2:
                add 1 to {_freeCount}

        set {_queueCount} to size of {bridge::queue::%{_arena}%::*}

        # Item de representação da Arena
        set {_item} to new ItemStack(Material.STAINED_CLAY, 1, 5) # Argila Verde
        set {_meta} to {_item}.getItemMeta()
        {_meta}.setDisplayName(colored "&a&lArena %{_arena}%")
        
        set {_lore} to Arrays.asList(colored "&7Clique para entrar na fila desta arena.", colored "", colored "&fInstâncias Livres: &a%{_freeCount}%", colored "&fJogadores na Fila: &e%{_queueCount}%", colored "", colored "&e▶ Clique para Jogar!")
        
        {_meta}.setLore({_lore})
        {_item}.setItemMeta({_meta})

        {_inv}.setItem({_slot}, {_item})
        add 1 to {_slot}
        if {_slot} is 17:
            exit loop

    {_p}.openInventory({_inv})

on inventory click:
    if name of event-inventory contains "The Bridge - Seleção de Arenas":
        cancel event
        if event-item is set:
            if event-item is not air:
                set {_meta} to event-item.getItemMeta()
                if {_meta} is set:
                    set {_displayName} to {_meta}.getDisplayName()
                    if {_displayName} contains "Arena ":
                        set {_arenaName} to uncolored {_displayName}
                        replace all "Arena " in {_arenaName} with ""
                        close player's inventory
                        execute player command "/jogar %{_arenaName}%"
