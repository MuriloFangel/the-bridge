on load:

    set {bridge::config::prefix} to "&9&lBRIDGE &8»&7"
    set {bridge::config::erro} to "&c&lERRO &8»&7"
    set {bridge::config::sucesso} to "&a&lSUCESSO &8»&7"

    # Variáveis Padrões das Arenas
    set {bridge::config::default_gols} to 5
    set {bridge::config::default_y_void} to 83

    set {bridge::config::default_tempo} to 900 # 15 minutos em segundos
    set {bridge::config::default_limite_ponte} to 20
    set {bridge::config::default_limite_lateral} to 20

    send "%{bridge::config::prefix}% &aMódulo de Configuração carregado!" to console
